## Metriche della tab Salute (definizioni precise)

| Metrica | Definizione | Soglia "ok" |
|---|---|---|
| **Sessioni** | Sequenza continua di campioni con gap ≤ 5 min | — |
| **Pausa** | Gap tra due sessioni > 5 min | ≥ 15 min = pausa vera |
| **Sessione lunga** | Sessione > 90 min | avviso |
| **Uso notturno** | Campioni tra 23:00 e 07:00 | 0 |
| **Primo utilizzo / ultimo** | primo/ultimo ts del giorno | — |
| **Pause lunghe** | n° pause ≥ 15 min | ≥ 3/giorno |
| **Streak igiene** | giorni consecutivi senza sessione >120 min E senza uso post-23:00 (ack giornaliero automatico se le condizioni sono rispettate alle 24:00) | crescente |

## KPI cards (4)

1. **Sessione più lunga oggi** — valore + badge ok/avviso (>90m)
2. **Pause fatte oggi** — n° pause ≥15min ("3 pause · la più lunga 45 min")
3. **Uso serale** — minuti dopo le 23:00 ("0 min · bene")
4. **Streak benessere** — giorni consecutivi "puliti"

## Sezioni pagina (tab Salute)

1. Timeline orizzontale del giorno: blocchi colore = sessioni, spazi = pause (SVG, come TinySpark ma full-width)
2. Lista sessioni di oggi: inizio–fine, durata, categoria dominante, flag ⚠ se >90m
3. Trend 7 giorni: barre delle sessioni >90m per giorno (0 = perfetto)
4. Nota metodologica footer (come le altre pagine)

```ts
// ============================================================
// ASCEND — Salute del PC: aggregazioni da campioni tracker
// Sessioni = sequenze continue (gap ≤ GAP_SESSION_MIN); pausa =
// gap > PAUSE_MIN_MIN. Notturno = campioni 23:00–07:00 locali.
// ============================================================
import type { TrackerSample } from "./pc-tracker";

export const GAP_SESSION_MIN = 5;   // ≤ questo gap = stessa sessione
export const LONG_SESSION_MIN = 90; // sessione lunga
const CRITICAL_SESSION_MIN = 120;   // rompe lo streak igiene
export const PAUSE_OK_MIN = 15;     // pausa che conta

export interface HealthSession {
  start: string; end: string;
  minutes: number;
  topCategory?: string; // opzionale, riempito dal chiamante col mapping
}

export interface DayHealth {
  date: string;
  firstUse: string | null;
  lastUse: string | null;
  totalMinutes: number;
  sessions: HealthSession[];
  longSessions: number;      // > LONG_SESSION_MIN
  longestSessionMin: number;
  pauses: { minutes: number }[];
  goodPauses: number;        // ≥ PAUSE_OK_MIN
  nightMinutes: number;      // 23:00–07:00
}

function parseTs(ts: string): number {
  const t = Date.parse(ts);
  return Number.isFinite(t) ? t : NaN;
}

/** Ora locale (0-24) di un ts ISO. */
export function hourLocal(iso: string): number {
  return new Date(iso).getHours();
}

export function aggregateDayHealth(
  dateKey: string,
  samples: TrackerSample[],
  tzOffsetGuard = true
): DayHealth {
  const valid = samples
    .filter((s) => s && typeof s.ts === "string")
    .map((s) => ({ ...s, _t: parseTs(s.ts) }))
    .filter((s) => !Number.isNaN(s._t))
    .sort((a, b) => a._t - b._t);

  const out: DayHealth = {
    date: dateKey,
    firstUse: valid[0]?.ts ?? null,
    lastUse: valid[valid.length - 1]?.ts ?? null,
    totalMinutes: 0,
    sessions: [],
    longSessions: 0,
    longestSessionMin: 0,
    pauses: [],
    goodPauses: 0,
    nightMinutes: 0,
  };
  if (valid.length === 0) return out;

  // sessioni per gap
  const GAP_MS = GAP_SESSION_MIN * 60_000;
  let segStart = valid[0]._t;
  let prev = valid[0]._t;
  const pushSeg = (endMs: number) => {
    const mins = Math.round(((endMs - segStart) / 60_000) * 10) / 10;
    // stima: ogni campione copre ~30s; il tempo "coperto" = end-start clampato ai gap interni
    out.sessions.push({
      start: new Date(segStart).toISOString(),
      end: new Date(endMs).toISOString(),
      minutes: Math.max(1, Math.round(mins)),
    });
  };
  for (let i = 1; i < valid.length; i++) {
    const t = valid[i]._t;
    const gap = t - prev;
    if (gap > GAP_MS) {
      pushSeg(prev);
      const pauseMin = Math.round(gap / 60_000);
      out.pauses.push({ minutes: pauseMin });
      segStart = t;
    }
    prev = t;
  }
  pushSeg(prev);

  for (const s of out.sessions) {
    out.longestSessionMin = Math.max(out.longestSessionMin, s.minutes);
    if (s.minutes > LONG_SESSION_MIN) out.longSessions++;
  }
  for (const p of out.pauses) {
    if (p.minutes >= PAUSE_OK_MIN) out.goodPauses++;
  }

  // totale = somma sessioni (esclude le pause)
  out.totalMinutes = out.sessions.reduce((acc, s) => acc + s.minutes, 0);

  // uso notturno: campioni tra 23:00 e 07:00 (ora locale)
  let night = 0;
  for (const s of valid) {
    const h = hourLocal(s.ts);
    if (h >= 23 || h < 7) night += 0.5; // ogni campione ≈ 30s
  }
  out.nightMinutes = Math.round(night);

  return out;
}

/** Streak "benessere": conta all'indietro dai giorni con ack salvati.
 *  Gli ack sono date-key yyyy-mm-dd salvate dal client quando il giorno
 *  finisce pulito (nessuna sessione >120m, nessun uso post-23). */
export function wellnessStreak(ackedDays: string[]): number {
  const set = new Set(ackedDays);
  let streak = 0;
  const d = new Date();
  for (;;) {
    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    if (!set.has(key)) break;
    streak++;
    d.setDate(d.getDate() - 1);
  }
  return streak;
}
```
